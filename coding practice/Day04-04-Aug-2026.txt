# Jenkinfiles - basic ci pipeline 
pipeline{
	agent any
	stages{
		stages('build'){
			steps{
				echo 'building project'
			}
		}
		stages('test'){
			steps {
				echo 'running tests'
		}
	}
	stage('deploy'){
	steps{
		echo 'deploying application'
		}
		}
	}
}


#  GitLab Ci/Cd — basic pipeline
stages:
  	- build
  	- test
  	- deploy

build:
  	stage: build
  	script:
    	- echo "building projects"

test:
  	stage: test
  	script:
    	- echo "running tests"

deploy:
  	stage: deploy
  	script:
    	- echo "deploying application"

# ansible — install and configure Nginx
- name: install and configure Nginx
hosts: all
become: yes
tasks:
    	- name: install nginx
     	yum:
        	name: nginx
        	state: present
    	- name: configure nginx
      	template:
        	src: nginx.conf.j2
        	dest: /etc/nginx/nginx.conf
    	- name: Start nginx service
      	service:
        	name: nginx
        	state: started
        	enabled: yes


# Terraform — ec2 with security group
resource "aws_security_group" "example" {
  	name = "example_security_group"
  	description = "allow SSH and HTTP"
  	ingress {
    		from_port = 22
    		to_port = 22
    		protocol = "tcp"
    		cidr_blocks = ["0.0.0.0/0"]
  	}

  	ingress {
    		from_port = 80
    		to_port = 80
    		protocol = "tcp"
    		cidr_blocks = ["0.0.0.0/0"]
  	}

  egress {
    		from_port = 0
    		to_port = 0
    		protocol = "-1"
    		cidr_blocks = ["0.0.0.0/0"]
  	}
}
