# kubernetes — horizontal pod auto scaler (HPA)
apiVersion: apps/v1
kind: Deployment
metadata:
  	name: pathnex-deployment
spec:
  	replicas: 1
  	selector:
    	matchLabels:
      	app: pathnex-app
  	template:
    		metadata:
      			labels:
        	app: pathnex-app
    	spec:
      	containers:
        	- name: nginx
          	image: nginx
          	ports:
            	- containerport: 80
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  	name: pathnex-hpa
spec:
  	scaleTargetRef:
    	apiVersion: apps/v1
    	kind: Deployment
    	name: pathnex-deployment
  	minReplicas: 1
  	maxReplicas: 10
  	targetCPUUtilizationPercentage: 50


# jenkinsfile — deploy to kubernetes
pipeline {
    	agent any
    	stages {
        	stage('Deploy to Kubernetes') {
            	steps {
                	script {
                    		sh 'kubectl apply -f deployment.yaml'
                		}
            		}
        	}
    	}
}
